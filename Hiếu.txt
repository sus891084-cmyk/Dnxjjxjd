local function saveConfig()
    if not writefile then return end
    pcall(function()
        writefile(CONFIG_FILE, HttpService:JSONEncode(config))
    end)
end

local function loadConfig()
    if not readfile or not isfile then return end
    pcall(function()
        if isfile(CONFIG_FILE) then
            local data = HttpService:JSONDecode(readfile(CONFIG_FILE))
            for k, v in pairs(data) do
                if config[k] ~= nil then
                    config[k] = v
                end
            end
        end
    end)
end

local function applyStats()
    local char = LocalPlayer.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then
            local spd = tonumber(SpeedBox.Text) or 16
            local jmp = tonumber(JumpBox.Text) or 50
            hum.WalkSpeed = spd
            pcall(function() hum.UseJumpPower = true end)
            hum.JumpPower = jmp
        end
    end
end

    btn.LayoutOrder = layoutOrder

    local btnCorner = Instance.new("UICorner", btn)
    btnCorner.CornerRadius = UDim.new(0, 6)

    local isToggled = config[configKey] or false

    local function updateVisuals()
        if isToggled then
            btn.BackgroundColor3 = Color3.fromRGB(50, 180, 50)
        else
            btn.BackgroundColor3 = Color3.fromRGB(45, 45, 45)

        isToggled = not isToggled
        config[configKey] = isToggled
        updateVisuals()
        saveConfig()
        if callback then callback(isToggled) end
    end)

    return btn
end

local function attackTarget(charFolder, targetObj)
    local char = charFolder:FindFirstChild(LocalPlayer.Name)
    if not char then return end
    
    local weapon = char:FindFirstChildOfClass("Tool")
    if not weapon then return end

    local hum = targetObj:FindFirstChildOfClass("Humanoid")
    local head = targetObj:FindFirstChild("Head")
    
    if hum and head and hum.Health > 0 then
        local player = Players:GetPlayerFromCharacter(targetObj)
        if player and player.Team == LocalPlayer.Team and not player.Neutral then return end

        local WeaponsSystem = ReplicatedStorage:FindFirstChild("WeaponsSystem")
        local Network = WeaponsSystem and WeaponsSystem:FindFirstChild("Network")
        local WeaponHit = Network and Network:FindFirstChild("WeaponHit")
        local WeaponFired = Network and Network:FindFirstChild("WeaponFired")
        
        local isMelee = not weapon:FindFirstChild("AmmoCapacity", true)

        pcall(function()
            if isMelee and WeaponFired then
                WeaponFired:FireServer(weapon, {
                    attackType = "Normal",
                    comboIndex = 1,
                    position = head.Position
                })
            elseif WeaponHit then
                WeaponHit:FireServer(weapon, {
                    p = head.Position,
                    pid = 1,
                    part = head,
                    d = 1,
                    maxDist = math.huge,
                    h = hum,
                    m = Enum.Material.Snow,
                    n = Vector3.new(1, 0, 0),
                    t = tick() % 1,
                    sid = math.random(1, 10)
                })
            end
        end)
    end
end

task.spawn(function()
    while task.wait(0.1) do
        if killAuraEnabled then
            local CharFolder = Workspace:FindFirstChild("Character")
            local myChar = CharFolder and CharFolder:FindFirstChild(LocalPlayer.Name)
            local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
            
            if CharFolder and myRoot then
                local maxDist = tonumber(AuraRadiusBox.Text) or 50
                
                for _, obj in pairs(CharFolder:GetChildren()) do
                    if obj:IsA("Model") and obj ~= myChar then
                        local hum = obj:FindFirstChildOfClass("Humanoid")
                        local root = obj:FindFirstChild("HumanoidRootPart") or obj:FindFirstChild("Head")
                        
                        if hum and hum.Health > 0 and root then
                            local dist = (root.Position - myRoot.Position).Magnitude
                            if dist <= maxDist then
                                attackTarget(CharFolder, obj)
                            end
                        end
                    end
                end
            end
        end
    end
end)

-- 6. LOCATION TRACKING (ЗЕЛЁНЫЕ МАРКЕРЫ)
-- =========================================================================
local markers = {}

local function createMarker(character)
    local head = character:FindFirstChild("Head")
    if not head then return nil end

    local highlight = Instance.new("Highlight")
    highlight.Name = "EnemyMarkerHL"
    highlight.FillColor = Color3.fromRGB(0, 255, 120)
    highlight.FillTransparency = 0.75
    highlight.OutlineColor = Color3.fromRGB(0, 255, 120)
    highlight.OutlineTransparency = 0
    highlight.Adornee = character
    highlight.Parent = character

    local billboard = Instance.new("BillboardGui")
    billboard.Name = "EnemyMarkerBB"
    billboard.Size = UDim2.new(0, 120, 0, 40)
    billboard.StudsOffset = Vector3.new(0, 3, 0)
    billboard.AlwaysOnTop = true
    billboard.Adornee = head
    billboard.Parent = character

    local distLabel = Instance.new("TextLabel")
    distLabel.Size = UDim2.new(1, 0, 1, 0)
    distLabel.BackgroundTransparency = 1
    distLabel.TextColor3 = Color3.fromRGB(0, 255, 120)
    distLabel.Font = Enum.Font.SourceSansBold
    distLabel.TextSize = 14
    distLabel.Text = "0m"
    distLabel.Parent = billboard

    return { highlight = highlight, billboard = billboard, distLabel = distLabel }
end

clearMarkers = function()
    for player, data in pairs(markers) do
        pcall(function()
            data.highlight:Destroy()
            data.billboard:Destroy()
        end)
        markers[player] = nil
    end
end

RunService.Heartbeat:Connect(function()
    if not trackEnabled then return end

    local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    local activePlayers = {}

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local char = player.Character
            local hum = char:FindFirstChildOfClass("Humanoid")
            local hrp = char:FindFirstChild("HumanoidRootPart")
            local head = char:FindFirstChild("Head")

            if hum and hum.Health > 0 and hrp and head then
                activePlayers[player] = true
                
                if not markers[player] then
                    local m = createMarker(char)
                    if m then markers[player] = m end
                else
                    local m = markers[player]
                    m.highlight.Adornee = char
                    m.billboard.Adornee = head
                    
                    if myRoot then
                        local dist = (hrp.Position - myRoot.Position).Magnitude
                        m.distLabel.Text = math.floor(dist) .. "m"
                    end
                end
            end
        end
    end

    for player, data in pairs(markers) do
        if not activePlayers[player] then
            pcall(function()
                data.highlight:Destroy()
                data.billboard:Destroy()
            end)
            markers[player] = nil
        end
    end
end)
